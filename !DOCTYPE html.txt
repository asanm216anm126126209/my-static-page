        function handleLogin(event) {
            event.preventDefault();
            
            const username = document.getElementById('username').value.trim();
            const password = document.getElementById('password').value.trim();
            const btn = document.getElementById('loginBtn');
            
            btn.textContent = '⏳ Checking...';
            btn.disabled = true;
            
            setTimeout(() => {
                const user = users.find(u => u.username === username && u.password === password);
                
                if (user) {
                    currentUser = user;
                    isLoggedIn = true;
                    
                    document.getElementById('loginPage').style.display = 'none';
                    document.getElementById('mainApp').style.display = 'block';
                    document.getElementById('userAvatar').textContent = user.name.charAt(0).toUpperCase();
                    document.getElementById('userNameDisplay').textContent = user.name;
                    
                    showToast(`✅ ${t('loginSuccess')} ${t('welcomeUser')}, ${user.name}!`, 'success');
                    
                    updateUItexts();
                    navigate('home');
                } else {
                    document.getElementById('loginForm').classList.add('animate-shake');
                    setTimeout(() => {
                        document.getElementById('loginForm').classList.remove('animate-shake');
                    }, 500);
                    showToast(`❌ ${t('loginFailed')}`, 'error');
                }
                
                btn.textContent = '🚀 Sign In';
                btn.disabled = false;
            }, 800);
        }

                document.getElementById('username').value = '';
                document.getElementById('password').value = '';
                
                showToast(`✅ ${t('logoutSuccess')}`, 'info');
            }
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            toast.className = `toast toast-${type}`;
            toast.textContent = message;
            container.appendChild(toast);

        // ============================================================
        // NAVIGATION
        // ============================================================
        function navigate(page) {
            currentPage = page;
            document.querySelectorAll('[data-page]').forEach(el => {
                el.classList.remove('active');
                if (el.dataset.page === page) el.classList.add('active');
            });
            document.getElementById('mobileMenu')?.classList.add('hidden');

            const content = document.getElementById('pageContent');
            switch (page) {
                case 'home': content.innerHTML = renderHome(); break;
                case 'dashboard': content.innerHTML = renderDashboard(); break;
                case 'patients': content.innerHTML = renderPatients(); break;
                case 'appointments': content.innerHTML = renderAppointments(); break;
                case 'doctors': content.innerHTML = renderDoctors(); break;
                case 'chatbot': content.innerHTML = renderChatbot(); break;
                default: content.innerHTML = renderHome();
            }
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function toggleMobile() {
            document.getElementById('mobileMenu').classList.toggle('hidden');
        }

        function deleteItem(type, id) {
            if (!confirm(`${t('confirmDelete')} ${type}?`)) return;

            let array;
            let name;

            switch (type) {
                case 'patient':
                    array = patients;
                    name = patients.find(i => i.id === id)?.name || '';
                    break;
                case 'appointment':
                    array = appointments;
                    name = appointments.find(i => i.id === id)?.patient || '';
                    break;
                case 'doctor':
                    array = doctors;
                    name = doctors.find(i => i.id === id)?.name || '';
                    break;
            }

            const index = array.findIndex(i => i.id === id);
            if (index > -1) {
                array.splice(index, 1);
                showToast(`🗑️ ${type} "${name}" ${t('deleteSuccess')}`, 'info');
                updateUI();
            }
        }

        function addChatMessage(type, text) {
            chatHistory.push({ type, text });
            const container = document.getElementById('chatMessages');
            if (!container) return;

            div.className = `flex ${type === 'user' ? 'justify-end' : 'justify-start'} animate-slide`;

        function processAICommand(message) {
            const lower = message.toLowerCase();

            if (lower.includes('help') || lower.includes('what can you do')) {
                addChatMessage('bot', `🤖 **I can help you with:**

        📋 **Patient Management**
        • Add patient: "Add patient John Doe, age 30, male, with flu"
        • Show patients: "Show all patients"

        📅 **Appointments**
        • Book: "Book appointment for John Smith with Dr. Williams tomorrow at 10 AM"
        • Show: "Show all appointments"

        👨‍⚕️ **Doctors**
        • Show: "Show all doctors"

        📊 **Statistics**
        • "Show dashboard" or "Stats"

        🎤 **Voice Commands**
        • Click the 🎤 button and speak naturally!`);
                return;
            }

            if (lower.includes('add patient') || lower.includes('new patient')) {
                const nameMatch = message.match(/add patient\s+([^,]+)/i) || message.match(/new patient\s+([^,]+)/i);
                const ageMatch = message.match(/age\s*(\d+)/i);
                const genderMatch = message.match(/\b(male|female|other)\b/i);
                const conditionMatch = message.match(/with\s+([^.]+)/i) || message.match(/for\s+([^.]+)/i);

                if (nameMatch) {
                    const name = nameMatch[1].trim();
                    const age = ageMatch ? parseInt(ageMatch[1]) : Math.floor(Math.random() * 50) + 20;
                    const gender = genderMatch ? genderMatch[1].charAt(0).toUpperCase() + genderMatch[1].slice(1) : 'Male';
                    const condition = conditionMatch ? conditionMatch[1].trim() : 'General Checkup';

                    patients.push({
                        id: patients.length + 1,
                        name: name,
                        age: age,
                        gender: gender,
                        condition: condition,
                        phone: '+1 234-567-' + String(9000 + patients.length).padStart(4, '0'),
                        status: 'Active'
                    });
                    addChatMessage('bot', `✅ **${t('patient')} "${name}" ${t('addSuccess')}**\n${t('age')}: ${age}, ${t('gender')}: ${gender}, ${t('condition')}: ${condition}`);
                    showToast(`✅ ${t('patient')} "${name}" ${t('addSuccess')}`, 'success');
                    updateUI();
                } else {
                    addChatMessage('bot', `📋 I need the patient name. Try: "Add patient John Doe, age 30, male, with flu"`);
                }
                return;
            }

            if (lower.includes('book appointment') || lower.includes('schedule appointment')) {
                const patientMatch = message.match(/for\s+([^,]+)/i) || message.match(/with patient\s+([^,]+)/i);
                const doctorMatch = message.match(/with\s+(dr\.?\s*\w+)/i);
                const dateMatch = message.match(/on\s+(\d{4}-\d{2}-\d{2})/i) || message.match(/tomorrow/i) || message.match(/today/i);
                const timeMatch = message.match(/at\s+(\d{1,2}\s*[AP]M)/i);

                if (patientMatch && doctorMatch) {
                    const patient = patientMatch[1].trim();
                    const doctor = doctorMatch[1].trim();
                    let date = '2024-01-20';
                    let time = '10:00 AM';

                    if (dateMatch) {
                        if (dateMatch[1]) date = dateMatch[1];
                        else if (dateMatch[0].includes('tomorrow')) {
                            const d = new Date();
                            d.setDate(d.getDate() + 1);
                            date = d.toISOString().split('T')[0];
                        }
                    }
                    if (timeMatch) time = timeMatch[1];

                    appointments.push({
                        id: appointments.length + 1,
                        patient: patient,
                        doctor: doctor,
                        date: date,
                        time: time,
                        status: 'Pending'

        let isListening = false;
        let recognition = null;

        function toggleVoice() {
            if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
                showToast('🎤 Voice recognition requires Chrome or Edge browser.', 'error');
                return;
            }

            const btn = document.getElementById('voiceBtn');
            const status = document.getElementById('voiceStatus');

            if (isListening) {
                stopVoice();
                return;
            }

            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            recognition = new SpeechRecognition();
            recognition.lang = 'en-US';
            recognition.continuous = false;
            recognition.interimResults = false;

            recognition.onstart = () => {
                isListening = true;
                btn.classList.add('listening');
                btn.textContent = '⏹️';
                status.classList.remove('hidden');
                status.textContent = '🎤 Listening... Speak now';
            };

            recognition.onresult = (event) => {
                const transcript = event.results[0][0].transcript;
                status.textContent = `🎤 You said: "${transcript}"`;
                document.getElementById('chatInput').value = transcript;
                speakText(`I heard you say: ${transcript}`);
                setTimeout(() => sendMessage(), 500);
                stopVoice();
            };

            recognition.onerror = (event) => {
                status.textContent = `❌ Error: ${event.error}`;
                setTimeout(() => stopVoice(), 2000);
            };

            recognition.onend = () => {
                stopVoice();
            };

            recognition.start();
        }

        function stopVoice() {
            if (recognition) {
                try { recognition.stop(); } catch (e) {}
            }
            isListening = false;
            const btn = document.getElementById('voiceBtn');
            const status = document.getElementById('voiceStatus');
            btn.classList.remove('listening');
            btn.textContent = '🎤';
            status.classList.add('hidden');
        }

        function startVoiceCommand() {
            if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
                showToast('🎤 Voice recognition requires Chrome or Edge browser.', 'error');
                return;
            }
            navigate('chatbot');
            setTimeout(() => toggleVoice(), 500);
        }

        function speakText(text) {
            if (!('speechSynthesis' in window)) return;
            
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.rate = 0.9;
            utterance.pitch = 1;
            utterance.volume = 1;
            utterance.lang = 'en-US';
            
            const voices = window.speechSynthesis.getVoices();
            const femaleVoice = voices.find(v => v.lang.startsWith('en') && v.name.includes('Female'));
            if (femaleVoice) utterance.voice = femaleVoice;
            
            window.speechSynthesis.speak(utterance);
        }

        if ('speechSynthesis' in window) {
            window.speechSynthesis.getVoices();
            window.speechSynthesis.onvoiceschanged = () => {
                window.speechSynthesis.getVoices();
            };
        }
